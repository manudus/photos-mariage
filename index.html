const FOLDER_NAME = "Mariage Valérie & Manu"; const INVITE_CODE = "6626"; /* ========================= 📁 RÉCUPÉRATION DOSSIER ========================= */ function getFolder() { const folders = DriveApp.getFoldersByName(FOLDER_NAME); return folders.hasNext() ? folders.next() : DriveApp.createFolder(FOLDER_NAME); } /* ========================= 📤 UPLOAD PHOTO (POST) ========================= */ function doPost(e) { try { const data = JSON.parse(e.postData.contents); // 🔐 sécurité simple if (!data.code || data.code !== INVITE_CODE) { return json({ success: false, error: "Code invalide" }); } const folder = getFolder(); const base64 = data.image.split(",")[1]; const blob = Utilities.newBlob( Utilities.base64Decode(base64), "image/jpeg", data.name || "photo.jpg" ); const file = folder.createFile(blob); // 🌍 rendre public file.setSharing( DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW ); const url = "https://drive.google.com/uc?export=view&id=" + file.getId(); // 💾 stockage liste images const props = PropertiesService.getScriptProperties(); let images = JSON.parse(props.getProperty("images") || "[]"); images.unshift(url); // plus récent en premier props.setProperty("images", JSON.stringify(images)); return json({ success: true, url: url }); } catch (err) { return json({ success: false, error: err.toString() }); } } /* ========================= 📥 GALERIE (GET) ========================= */ function doGet() { const props = PropertiesService.getScriptProperties(); const images = JSON.parse(props.getProperty("images") || "[]"); return json({ success: true, images: images }); } /* ========================= 🔧 UTILITAIRE JSON ========================= */ function json(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
